**Goal:** Add an Atto-backed nested-value parsing path for canonical `build_model` while preserving the public `build_model` API and model semantics.

**Architecture:** Add `atto` to the core `ccl` package and create a focused internal module, `ccl/model/nested_parser.gleam`, that uses Atto combinators to recognize structurally nested model values before delegating to the existing CCL parser for parse-option semantics. `src/ccl/model.gleam` keeps folding and merging models, but calls the new helper instead of directly checking string prefixes and parsing nested values.

```gleam
@internal
import atto
import atto/ops
import atto/text
import ccl/parser
import ccl/types.{type Entry, type ParseOptions}
import gleam/result
import gleam/string

pub fn parse(
  raw_value: String,
  parse_options: ParseOptions,
) -> Result(List(Entry), Nil) {
  use nested_source <- result.try(
    atto.run(nested_value_source(), text.new(raw_value), Nil)
    |> result.map_error(fn(_) { Nil }),
  )

  parser.parse_value_with(nested_source, parse_options)
  |> result.map_error(fn(_) { Nil })
}

fn nested_value_source() -> atto.Parser(String, String, String, Nil, Nil) {
  use first <- atto.do(ops.choice([atto.token("\r\n"), atto.token("\n")]))
  use rest <- atto.do(ops.many(atto.any()))
  use _ <- atto.do(atto.eof())

  atto.pure(first <> string.concat(rest))
}
```
